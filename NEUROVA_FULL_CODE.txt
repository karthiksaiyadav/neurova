  );
}

const supabaseUrl = import.meta.env.VITE_SUPABASE_URL;
const supabaseAnonKey = import.meta.env.VITE_SUPABASE_ANON_KEY;

export const supabase = createClient(supabaseUrl, supabaseAnonKey);

export interface SensorReading {
  id: number;
  device_id: string;
  temperature: number | null;
  humidity: number | null;
  vibration_amplitude: number | null;
  vibration_event: boolean;
  temperature_warning: boolean;
  humidity_warning: boolean;
  total_risk: number | null;
  risk_alert: boolean;
  created_at: string;
}

export const ALERT_THRESHOLDS = {
  tempWarning: 35.0,
  rhWarning: 65.0,
  rhHigh: 75.0,
  piezoEventThreshold: 250,
  riskAlertLimit: 30.0,
};

export function computeAlerts(r: SensorReading): SensorReading {
  const temperatureWarning =
    r.temperature !== null && r.temperature > ALERT_THRESHOLDS.tempWarning;
  const humidityWarning =
    r.humidity !== null && r.humidity > ALERT_THRESHOLDS.rhWarning;
  const vibrationEvent =
    r.vibration_event ||
    (r.vibration_amplitude !== null && r.vibration_amplitude >= ALERT_THRESHOLDS.piezoEventThreshold);
  const riskAlert =
    r.risk_alert ||
    (r.total_risk !== null && r.total_risk > ALERT_THRESHOLDS.riskAlertLimit);

  return {
    ...r,
    temperature_warning: temperatureWarning,
    humidity_warning: humidityWarning,
    vibration_event: vibrationEvent,
    risk_alert: riskAlert,
  };
}

import { createContext, useCallback, useContext, useMemo, useState, type ReactNode } from "react";
import type { GrainColumn, NormalizeResult } from "@/lib/grain/csv";
import { generateSampleData } from "@/lib/grain/sample";
import type { OverallScore } from "@/lib/grain/scoring";
import { scoreDataset } from "@/lib/grain/scoring";

interface DatasetState {
  rows: NormalizeResult["rows"];
  mapping: Record<string, GrainColumn | "">;
  fileName: string | null;
  stats: NormalizeResult["stats"];
  isLoaded: boolean;
  isSample: boolean;
}

interface DatasetContextValue extends DatasetState {
  loadCsv: (data: { rows: NormalizeResult["rows"]; mapping: Record<string, GrainColumn | "">; fileName: string; stats: NormalizeResult["stats"] }) => void;
  loadSample: () => void;
  clear: () => void;
  score: OverallScore | null;
}

const DatasetContext = createContext<DatasetContextValue | null>(null);

const emptyState: DatasetState = {
  rows: [],
  mapping: {},
  fileName: null,
  stats: {},
  isLoaded: false,
  isSample: false,
};

export function DatasetProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<DatasetState>(emptyState);

  const loadCsv = useCallback((data: { rows: NormalizeResult["rows"]; mapping: Record<string, GrainColumn | "">; fileName: string; stats: NormalizeResult["stats"] }) => {
    setState({
      rows: data.rows,
      mapping: data.mapping,
      fileName: data.fileName,
      stats: data.stats,
      isLoaded: true,
      isSample: false,
    });
  }, []);

  const loadSample = useCallback(() => {
    const rows = generateSampleData();
    setState({
      rows,
      mapping: {},
      fileName: "sample-dataset.csv",
      stats: {},
      isLoaded: true,
      isSample: true,
    });
  }, []);

  const clear = useCallback(() => setState(emptyState), []);

  const score = useMemo(() => (state.rows.length > 0 ? scoreDataset(state.rows) : null), [state.rows]);

  const value = useMemo<DatasetContextValue>(
    () => ({ ...state, loadCsv, loadSample, clear, score }),
    [state, loadCsv, loadSample, clear, score],
  );

  return <DatasetContext.Provider value={value}>{children}</DatasetContext.Provider>;
}

export function useDataset() {
  const ctx = useContext(DatasetContext);
  if (!ctx) throw new Error("useDataset must be used within a DatasetProvider");
  return ctx;
}

export type GrainColumn =
  | "timestamp"
  | "sensor_id"
  | "temperature"
  | "humidity";

export const COLUMN_LABELS: Record<GrainColumn, string> = {
  timestamp: "Timestamp",
  sensor_id: "Sensor / Bin ID",
  temperature: "Temperature (°C)",
  humidity: "Ambient Humidity (%)",
};

export const COLUMN_ALIASES: Record<GrainColumn, string[]> = {
  timestamp: ["timestamp", "time", "date", "datetime", "reading_time", "recorded_at", "ts"],
  sensor_id: ["sensor_id", "sensor", "bin_id", "bin", "location", "silo", "probe", "node", "device_id"],
  temperature: ["temperature", "temp", "temperature_c", "temp_c", "grain_temp", "probe_temp"],
  humidity: ["humidity", "ambient_humidity", "rh", "relative_humidity", "air_humidity"],
};

export const ALL_COLUMNS: GrainColumn[] = ["timestamp", "sensor_id", "temperature", "humidity"];

export const REQUIRED_COLUMNS: GrainColumn[] = ["timestamp", "sensor_id", "temperature"];

export interface CsvParseResult {
  headers: string[];
  rows: string[][];
}

export interface ValidationIssue {
  level: "error" | "warning";
  message: string;
  row?: number;
  column?: GrainColumn;
}

export interface ColumnStats {
  count: number;
  min: number | null;
  max: number | null;
  mean: number | null;
  unit: string;
}

export interface NormalizedRow {
  timestamp: string;
  sensor_id: string;
  temperature: number | null;
  humidity: number | null;
}

export interface NormalizeResult {
  rows: NormalizedRow[];
  issues: ValidationIssue[];
  stats: Partial<Record<GrainColumn, ColumnStats>>;
  missing: Partial<Record<GrainColumn, number>>;
}

function detectDelimiter(text: string): string {
  const firstLine = text.split(/\r?\n/)[0] ?? "";
  const candidates = [",", "\t", ";", "|"];
  let best = ",";
  let bestCount = 0;
  for (const d of candidates) {
    const count = firstLine.split(d).length - 1;
    if (count > bestCount) {
      best = d;
      bestCount = count;
    }
  }
  return best;
}

function parseLine(line: string, delimiter: string): string[] {
  const result: string[] = [];
  let current = "";
  let inQuotes = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (inQuotes) {
      if (ch === '"') {
        if (line[i + 1] === '"') {
          current += '"';
          i++;
        } else {
          inQuotes = false;
        }
      } else {
        current += ch;
      }
    } else {
      if (ch === '"') {
        inQuotes = true;
      } else if (ch === delimiter) {
        result.push(current.trim());
        current = "";
      } else {
        current += ch;
      }
    }
  }
  result.push(current.trim());
  return result;
}

export function parseCsv(text: string): CsvParseResult {
  const delimiter = detectDelimiter(text);
  const lines = text.split(/\r?\n/).filter((l) => l.trim().length > 0);
  if (lines.length === 0) return { headers: [], rows: [] };
  const headers = parseLine(lines[0], delimiter);
  const rows = lines.slice(1).map((l) => parseLine(l, delimiter));
  return { headers, rows };
}

export function normalizeHeader(h: string): string {
  return h.toLowerCase().replace(/[\s\-._]/g, "_").replace(/[^a-z0-9_]/g, "");
}

export function autoMap(headers: string[]): Record<string, GrainColumn | ""> {
  const mapping: Record<string, GrainColumn | ""> = {};
  const used = new Set<GrainColumn>();

  for (const header of headers) {
    const normalized = normalizeHeader(header);
    let matched: GrainColumn | "" = "";

    for (const col of ALL_COLUMNS) {
      if (used.has(col)) continue;
      const aliases = COLUMN_ALIASES[col].map(normalizeHeader);
      if (aliases.includes(normalized)) {
        matched = col;
        break;
      }
    }

    if (!matched) {
      for (const col of ALL_COLUMNS) {
        if (used.has(col)) continue;
        const aliases = COLUMN_ALIASES[col].map(normalizeHeader);
        if (aliases.some((a) => normalized.includes(a) || a.includes(normalized))) {
          matched = col;
          break;
        }
      }
    }

    if (matched) used.add(matched);
    mapping[header] = matched;
  }

  return mapping;
}

function parseNumber(value: string): number | null {
  if (!value || value.trim() === "") return null;
  const cleaned = value.replace(/[^0-9.\-+eE]/g, "");
  const n = parseFloat(cleaned);
  return isNaN(n) ? null : n;
}

const COLUMN_UNITS: Record<GrainColumn, string> = {
  timestamp: "",
  sensor_id: "",
  temperature: "°C",
  humidity: "%",
};

const COLUMN_RANGES: Partial<Record<GrainColumn, { min: number; max: number }>> = {
  temperature: { min: -40, max: 80 },
  humidity: { min: 0, max: 100 },
};

function computeStats(values: number[]): ColumnStats {
  const valid = values.filter((v) => v !== null && !isNaN(v)) as number[];
  if (valid.length === 0) {
    return { count: 0, min: null, max: null, mean: null, unit: "" };
  }
  const sum = valid.reduce((a, b) => a + b, 0);
  const col = Object.keys(COLUMN_RANGES).find(() => true);
  return {
    count: valid.length,
    min: Math.min(...valid),
    max: Math.max(...valid),
    mean: sum / valid.length,
    unit: col ? COLUMN_UNITS[col as GrainColumn] : "",
  };
}

export function normalizeRows(
  rows: string[][],
  headers: string[],
  mapping: Record<string, GrainColumn | "">,
): NormalizeResult {
  const issues: ValidationIssue[] = [];
  const missing: Partial<Record<GrainColumn, number>> = {};
  const numericValues: Partial<Record<GrainColumn, number[]>> = {};

  for (const col of ALL_COLUMNS) {
    missing[col] = 0;
    numericValues[col] = [];
  }

  for (const required of REQUIRED_COLUMNS) {
    const hasMapping = Object.values(mapping).includes(required);
    if (!hasMapping) {
      issues.push({
        level: "error",
        message: `Required column "${COLUMN_LABELS[required]}" is not mapped. Assign it in the mapping above.`,
        column: required,
      });
    }
  }

  const normalizedRows: NormalizedRow[] = rows.map((rawRow, rowIdx) => {
    const rowMap: Record<string, string> = {};
    headers.forEach((h, i) => {
      rowMap[h] = rawRow[i] ?? "";
    });

    const get = (col: GrainColumn): string => {
      const header = Object.entries(mapping).find(([, v]) => v === col)?.[0];
      return header ? rowMap[header] ?? "" : "";
    };

    const timestamp = get("timestamp");
    const sensor_id = get("sensor_id");
    const tempStr = get("temperature");
    const humidStr = get("humidity");

    if (!timestamp) missing.timestamp!++;
    if (!sensor_id) missing.sensor_id!++;
    if (!tempStr) missing.temperature!++;

    const temperature = parseNumber(tempStr);
    const humidity = parseNumber(humidStr);

    if (temperature !== null) numericValues.temperature!.push(temperature);
    if (humidity !== null) numericValues.humidity!.push(humidity);

    const rowNum = rowIdx + 1;

    for (const [col, value] of [
      ["temperature", temperature],
      ["humidity", humidity],
    ] as [GrainColumn, number | null][]) {
      const range = COLUMN_RANGES[col];
      if (range && value !== null) {
        if (value < range.min || value > range.max) {
          issues.push({
            level: "warning",
            message: `Row ${rowNum}: ${COLUMN_LABELS[col]} value ${value}${COLUMN_UNITS[col]} is outside the expected range (${range.min}–${range.max}).`,
            row: rowNum,
            column: col,
          });
        }
      }
    }

    if (REQUIRED_COLUMNS.includes("timestamp") && !timestamp) {
      issues.push({
        level: "error",
        message: `Row ${rowNum}: Missing timestamp.`,
        row: rowNum,
        column: "timestamp",
      });
    }

    if (REQUIRED_COLUMNS.includes("sensor_id") && !sensor_id) {
      issues.push({
        level: "warning",
        message: `Row ${rowNum}: Missing sensor/bin ID.`,
        row: rowNum,
        column: "sensor_id",
      });
    }

    return { timestamp, sensor_id, temperature, humidity };
  });

  const stats: Partial<Record<GrainColumn, ColumnStats>> = {};
  for (const col of ["temperature", "humidity"] as GrainColumn[]) {
    stats[col] = {
      ...computeStats(numericValues[col]!),
      unit: COLUMN_UNITS[col],
    };
  }

  return { rows: normalizedRows, issues, stats, missing };
}

  maxTemp: number | null;
  maxHumidity: number | null;
  trend: "stable" | "rising" | "falling";
}

export interface OverallScore {
  overall: number;
  bins: BinScore[];
  totalReadings: number;
  safeBins: number;
  cautionBins: number;
  warningBins: number;
  criticalBins: number;
  worstBin: BinScore | null;
}

function scoreTemperature(avg: number | null, max: number | null): number {
  if (avg === null) return 50;
  let score = 100;
  if (avg > 15) score -= (avg - 15) * 4;
  if (avg > 25) score -= (avg - 25) * 6;
  if (max !== null && max > 30) score -= (max - 30) * 3;
  return Math.max(0, Math.min(100, score));
}

function scoreHumidity(avg: number | null, max: number | null): number {
  if (avg === null) return 50;
  let score = 100;
  if (avg > 55) score -= (avg - 55) * 1.5;
  if (avg > 65) score -= (avg - 65) * 2.5;
  if (avg > 75) score -= (avg - 75) * 3;
  if (max !== null && max > 80) score -= (max - 80) * 2;
  return Math.max(0, Math.min(100, score));
}

function riskFromScore(score: number): "safe" | "caution" | "warning" | "critical" {
  if (score >= 80) return "safe";
  if (score >= 60) return "caution";
  if (score >= 40) return "warning";
  return "critical";
}

function detectTrend(rows: NormalizedRow[]): "stable" | "rising" | "falling" {
  if (rows.length < 4) return "stable";
  const quarter = Math.floor(rows.length / 4);
  const first = rows.slice(0, quarter);
  const last = rows.slice(-quarter);
  const avgFirst = avg(first.map((r) => r.temperature).filter((v): v is number => v !== null));
  const avgLast = avg(last.map((r) => r.temperature).filter((v): v is number => v !== null));
  if (avgFirst === null || avgLast === null) return "stable";
  const diff = avgLast - avgFirst;
  if (diff > 1.5) return "rising";
  if (diff < -1.5) return "falling";
  return "stable";
}

function avg(nums: number[]): number | null {
  if (nums.length === 0) return null;
  return nums.reduce((a, b) => a + b, 0) / nums.length;
}

export function scoreDataset(rows: NormalizedRow[]): OverallScore {
  const byBin = new Map<string, NormalizedRow[]>();
  for (const row of rows) {
    if (!row.sensor_id) continue;
    const existing = byBin.get(row.sensor_id) ?? [];
    existing.push(row);
    byBin.set(row.sensor_id, existing);
  }

  const bins: BinScore[] = [];

  for (const [sensorId, binRows] of byBin) {
    const temps = binRows.map((r) => r.temperature).filter((v): v is number => v !== null);
    const humidities = binRows.map((r) => r.humidity).filter((v): v is number => v !== null);

    const avgTemp = avg(temps);
    const avgHumidity = avg(humidities);
    const maxTemp = temps.length ? Math.max(...temps) : null;
    const maxHumidity = humidities.length ? Math.max(...humidities) : null;

    const tempScore = scoreTemperature(avgTemp, maxTemp);
    const humidScore = scoreHumidity(avgHumidity, maxHumidity);

    const overall = Math.round((tempScore * 0.55 + humidScore * 0.45));

    bins.push({
      sensor_id: sensorId,
      overall,
      temperature: Math.round(tempScore),
      humidity: Math.round(humidScore),
      riskLevel: riskFromScore(overall),
      readingCount: binRows.length,
      latestReading: binRows[binRows.length - 1],
      avgTemp: avgTemp !== null ? parseFloat(avgTemp.toFixed(1)) : null,
      avgHumidity: avgHumidity !== null ? parseFloat(avgHumidity.toFixed(1)) : null,
      maxTemp: maxTemp !== null ? parseFloat(maxTemp.toFixed(1)) : null,
      maxHumidity: maxHumidity !== null ? parseFloat(maxHumidity.toFixed(1)) : null,
      trend: detectTrend(binRows),
    });
  }

  bins.sort((a, b) => a.overall - b.overall);

  const safeBins = bins.filter((b) => b.riskLevel === "safe").length;
  const cautionBins = bins.filter((b) => b.riskLevel === "caution").length;
  const warningBins = bins.filter((b) => b.riskLevel === "warning").length;
  const criticalBins = bins.filter((b) => b.riskLevel === "critical").length;

  const overallScore = bins.length > 0
    ? Math.round(bins.reduce((sum, b) => sum + b.overall, 0) / bins.length)
    : 0;

  return {
    overall: overallScore,
    bins,
    totalReadings: rows.length,
    safeBins,
    cautionBins,
    warningBins,
    criticalBins,
    worstBin: bins[0] ?? null,
  };
}

import type { NormalizedRow } from "./csv";
import { normalizeRows, autoMap, parseCsv } from "./csv";

export const SAMPLE_LABEL = "12 bins · 7 days · 10-min interval";
export const SAMPLE_NOTES = [
  "Twelve storage bins across two sites with realistic temperature and humidity drift.",
  "Readings every 10 minutes for 7 days — 12 096 rows in total.",
  "Two bins are deliberately degrading to show how the scoring flags risk early.",
  "Everything is generated in your browser, so the sample never touches a server.",
];

function seededRandom(seed: number): () => number {
  let s = seed;
  return () => {
    s = (s * 9301 + 49297) % 233280;
    return s / 233280;
  };
}

const BIN_NAMES = [
  "BIN-A1", "BIN-A2", "BIN-A3", "BIN-A4",
  "BIN-A5", "BIN-A6", "BIN-B1", "BIN-B2",
  "BIN-B3", "BIN-B4", "BIN-B5", "BIN-B6",
];

export function generateSampleData(): NormalizedRow[] {
  const rand = seededRandom(42);
  const rows: NormalizedRow[] = [];
  const startDate = new Date("2025-09-01T00:00:00Z");
  const intervalMin = 10;
  const totalDays = 7;
  const readingsPerBin = (totalDays * 24 * 60) / intervalMin;

  for (let b = 0; b < BIN_NAMES.length; b++) {
    const bin = BIN_NAMES[b];
    const baseTemp = 14 + rand() * 8;
    const baseHumidity = 45 + rand() * 20;

    const degrading = b === 4 || b === 9;

    for (let r = 0; r < readingsPerBin; r++) {
      const ts = new Date(startDate.getTime() + r * intervalMin * 60 * 1000);
      const dayProgress = r / readingsPerBin;

      const diurnal = Math.sin((r / readingsPerBin) * Math.PI * 2 * 7) * 2;

      let temp = baseTemp + diurnal + (rand() - 0.5) * 1.5;
      let humidity = baseHumidity + (rand() - 0.5) * 5;

      if (degrading) {
        temp += dayProgress * 16;
        humidity += dayProgress * 18;
      }

      rows.push({
        timestamp: ts.toISOString(),
        sensor_id: bin,
        temperature: parseFloat(temp.toFixed(1)),
        humidity: parseFloat(humidity.toFixed(1)),
      });
    }
  }

  return rows;
}

export function generateSampleCsv(): string {
  const rows = generateSampleData();
  const headers = ["timestamp", "sensor_id", "temperature", "humidity"];
  const lines = [headers.join(",")];
  for (const row of rows) {
    lines.push([row.timestamp, row.sensor_id, row.temperature, row.humidity].join(","));
  }
  return lines.join("\n");
}

export function loadSampleAsParsed() {
  const csv = generateSampleCsv();
  const parsed = parseCsv(csv);
  const mapping = autoMap(parsed.headers);
  const normalized = normalizeRows(parsed.rows, parsed.headers, mapping);
  return { parsed, mapping, normalized };
}

import { ValidationReport } from "@/components/upload/ValidationReport";
import { Button } from "@/components/ui/button";
import { useDataset } from "@/lib/dataset/DatasetContext";
import { autoMap, normalizeRows, parseCsv, type GrainColumn } from "@/lib/grain/csv";
import { SAMPLE_LABEL, SAMPLE_NOTES } from "@/lib/grain/sample";

const MAX_BYTES = 20 * 1024 * 1024;

interface UploadPageProps {
  onNavigate: (page: string) => void;
}

export function UploadPage({ onNavigate }: UploadPageProps) {
  const { loadCsv, loadSample } = useDataset();
  const [parsed, setParsed] = useState<{ headers: string[]; rows: string[][]; fileName: string } | null>(null);
  const [mapping, setMapping] = useState<Record<string, GrainColumn | "">>({});
  const [error, setError] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);

  const normalized = useMemo(
    () => (parsed ? normalizeRows(parsed.rows, parsed.headers, mapping) : null),
    [parsed, mapping],
  );

  const blockers = normalized ? normalized.issues.filter((issue) => issue.level === "error") : [];

  const handleFile = async (file: File) => {
    setError(null);
    if (file.size > MAX_BYTES) {
      setError("That file is larger than the 20 MB limit.");
      return;
    }
    setBusy(true);
    try {
      const text = await file.text();
      const result = parseCsv(text);
      if (!result.headers.length || !result.rows.length) {
        setError("No header row and data rows could be found in that file.");
        return;
      }
      setParsed({ fileName: file.name, ...result });
      setMapping(autoMap(result.headers));
    } catch {
      setError("That file could not be read. A plain CSV, tab or semicolon delimited file works best.");
    } finally {
      setBusy(false);
    }
  };

  const handleLoad = () => {
    if (!normalized || blockers.length || !parsed) return;
    loadCsv({ rows: normalized.rows, mapping, fileName: parsed.fileName, stats: normalized.stats });
    onNavigate("dashboard");
  };

  const handleSample = () => {
    loadSample();
    onNavigate("dashboard");
  };

  return (
    <div className="mx-auto max-w-5xl">
      <PageHeader
        title="Upload CSV"
        subtitle="Bring in grain-storage readings from a logger export, or start from the built-in sample dataset."
      >
        <StatusPill />
      </PageHeader>

      <DatasetGate title="Nothing imported yet" description="Drop a CSV above to map its columns, or load the sample dataset.">
        <div className="space-y-4">
          <CsvDropzone onFile={handleFile} busy={busy} fileName={parsed?.fileName} />

          {error ? (
            <p className="rounded-xl border border-destructive bg-destructive-soft px-4 py-3 text-xs text-destructive-foreground">{error}</p>
          ) : null}

          {parsed && normalized ? (
            <>
              <SectionCard
                title="Column alias mapping"
                description="Header names are matched automatically. Adjust anything the file names differently — the mapping applies to every reading."
              >
                <ColumnMappingForm
                  headers={parsed.headers}
                  mapping={mapping}
                  onChange={(key, value) => setMapping((current) => ({ ...current, [key]: value }))}
                />
              </SectionCard>

              <SectionCard title="Validation & missing values" description="What the importer saw, including every value it did not receive.">
                <ValidationReport issues={normalized.issues} stats={normalized.stats} missing={normalized.missing} />
              </SectionCard>

              <SectionCard title="Preview" description="The first readings after mapping.">
                <DataPreviewTable rows={normalized.rows} />
              </SectionCard>

              <div className="flex flex-wrap items-center gap-3">
                <Button onClick={handleLoad} disabled={Boolean(blockers.length)}>
                  <ArrowRight className="h-4 w-4" />
                  Run the analysis
                </Button>
                <span className="text-[11px] text-muted-foreground">
                  {blockers.length ? blockers[0].message : "Scoring happens in this browser — nothing is uploaded to a server."}
                </span>
              </div>
            </>
          ) : null}

          <SectionCard
            title="No sensor handy?"
            description="The sample dataset is generated in your browser so every page of the lab can be demonstrated without hardware."
          >
            <ul className="space-y-2">
              {SAMPLE_NOTES.map((note) => (
                <li key={note} className="flex gap-2.5 text-xs text-muted-foreground">
                  <span className="mt-1.5 h-1.5 w-1.5 shrink-0 rounded-full bg-primary-500" />
                  {note}
                </li>
              ))}
            </ul>
            <div className="mt-4 flex flex-wrap items-center gap-3">
              <Button variant="outline" onClick={handleSample}>
                <DatabaseZap className="h-4 w-4" />
                Load Sample Dataset
              </Button>
              <span className="rounded-full border border-warning bg-warning-soft px-3 py-1 text-[10px] font-semibold uppercase tracking-wider text-warning-foreground">
                {SAMPLE_LABEL}
              </span>
            </div>
          </SectionCard>

          <p className="flex items-center gap-2 text-[11px] text-muted-foreground">
            <UploadCloud className="h-3.5 w-3.5" />
            Files are read locally with the browser's file reader. The dataset stays in this browser and never mixes with live
            or sample data.
          </p>
        </div>
      </DatasetGate>
    </div>
  );
}

interface BinsPageProps {
  onNavigate: (page: string) => void;
}

type SortKey = "sensor_id" | "overall" | "temperature" | "humidity";

export function BinsPage(_: BinsPageProps) {
  const { score } = useDataset();
  const [sortKey, setSortKey] = useState<SortKey>("overall");
  const [sortDir, setSortDir] = useState<"asc" | "desc">("asc");
  const [expanded, setExpanded] = useState<string | null>(null);

  if (!score) return null;

  const sorted = [...score.bins].sort((a, b) => {
    let cmp: number;
    if (sortKey === "sensor_id") {
      cmp = a.sensor_id.localeCompare(b.sensor_id);
    } else {
      cmp = (a[sortKey] as number) - (b[sortKey] as number);
    }
    return sortDir === "asc" ? cmp : -cmp;
  });

  const toggleSort = (key: SortKey) => {
    if (sortKey === key) {
      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
    } else {
      setSortKey(key);
      setSortDir("asc");
    }
  };

  const riskBadge: Record<string, string> = {
    safe: "bg-success-soft text-success-foreground border-success-200",
    caution: "bg-warning-soft text-warning-foreground border-warning-200",
    warning: "bg-warning-soft text-warning-foreground border-warning-300",
    critical: "bg-destructive-soft text-destructive-foreground border-destructive-200",
  };

  const trendBadge: Record<string, string> = {
    stable: "bg-secondary-100 text-secondary-600",
    rising: "bg-destructive-soft text-destructive-foreground",
    falling: "bg-success-soft text-success-foreground",
  };

  const SortHeader = ({ label, sortK }: { label: string; sortK: SortKey }) => (
    <th
      className="cursor-pointer select-none py-2 pr-4 font-medium transition-colors hover:text-secondary-700"
      onClick={() => toggleSort(sortK)}
    >
      <span className="inline-flex items-center gap-1">
        {label}
        {sortKey === sortK ? (
          sortDir === "asc" ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />
        ) : null}
      </span>
    </th>
  );

  return (
    <div className="mx-auto max-w-6xl">
      <PageHeader title="Bins" subtitle="Per-bin breakdown of every sensor reading and risk score">
        <StatusPill />
      </PageHeader>

      <div className="overflow-x-auto rounded-2xl border border-secondary-200 bg-white shadow-sm scrollbar-thin">
        <table className="w-full text-left text-xs">
          <thead className="border-b border-secondary-200 bg-secondary-50 text-secondary-500">

              <SortHeader label="Sensor" sortK="sensor_id" />
              <SortHeader label="Overall" sortK="overall" />
              <SortHeader label="Temp" sortK="temperature" />
              <SortHeader label="Humidity" sortK="humidity" />

    return bins.slice(0, 6);

    const byBin = new Map<string, { ts: Date; value: number }[]>();
    for (const row of rows) {
      if (!row.sensor_id) continue;
      const value = row[metric];
      if (value === null) continue;
      const arr = byBin.get(row.sensor_id) ?? [];
      arr.push({ ts: new Date(row.timestamp), value });
      byBin.set(row.sensor_id, arr);

    const vals: number[] = [];

      const data = chartData.get(bin);
      if (data) vals.push(...data.map((d) => d.value));
    }
    return vals;
  }, [chartData, displayedBins]);

  const minVal = allValues.length ? Math.min(...allValues) : 0;
  const maxVal = allValues.length ? Math.max(...allValues) : 100;
  const valRange = maxVal - minVal || 1;

  const allTimes = useMemo(() => {
    const times: Date[] = [];
    for (const bin of displayedBins) {
      const data = chartData.get(bin);
      if (data) times.push(...data.map((d) => d.ts));
    }
    return times.sort((a, b) => a.getTime() - b.getTime());
  }, [chartData, displayedBins]);

  const minTime = allTimes.length ? allTimes[0].getTime() : 0;
  const maxTime = allTimes.length ? allTimes[allTimes.length - 1].getTime() : 1;

                })
                .join(" ");

function getActiveAlerts(latest: SensorReading | null): Array<{ severity: "critical" | "warning"; title: string; detail: string }> {
  if (!latest) return [];
  const alerts: Array<{ severity: "critical" | "warning"; title: string; detail: string }> = [];

  if (latest.risk_alert && latest.total_risk !== null) {
    alerts.push({
      severity: "critical",
      title: `Risk alert active — score ${latest.total_risk.toFixed(1)}`,
      detail: `Combined risk exceeded ${RISK_ALERT_LIMIT}. Check temperature, humidity, and vibration immediately.`,
    });
  }

  if (latest.temperature_warning && latest.temperature !== null) {
    alerts.push({
      severity: "critical",
      title: `Temperature high — ${latest.temperature.toFixed(1)}°C`,
      detail: `Above warning threshold of ${TEMP_WARNING}°C. Grain storage temperature is too high.`,
    });
  }

  if (latest.humidity !== null && latest.humidity > RH_HIGH) {
    alerts.push({
      severity: "critical",
      title: `Humidity critical — ${latest.humidity.toFixed(1)}%`,
      detail: `Above high threshold of ${RH_HIGH}%. Moisture condensation and mold growth likely.`,
    });
  } else if (latest.humidity_warning && latest.humidity !== null) {
    alerts.push({
      severity: "warning",
      title: `Humidity elevated — ${latest.humidity.toFixed(1)}%`,
      detail: `Above warning threshold of ${RH_WARNING}%. Monitor closely for spoilage indicators.`,
    });
  }

  if (latest.vibration_event && latest.vibration_amplitude !== null) {
    alerts.push({
      severity: "warning",
      title: `Vibration event detected — amplitude ${latest.vibration_amplitude}`,
      detail: `Piezo amplitude crossed ${PIEZO_EVENT_THRESHOLD} ADC. Possible impact or pest activity.`,
    });
  }

  return alerts;
}

function MiniChart({
  values,
  color,
  unit,
  max,
  threshold,
  thresholdLabel,
}: {
  values: number[];
  color: string;
  unit: string;
  max?: number;
  threshold?: number;
  thresholdLabel?: string;
}) {
  if (values.length < 2) {
    return <p className="py-8 text-center text-xs text-secondary-400">Not enough data yet.</p>;
  }

  const minVal = Math.min(...values);
  const maxVal = max ?? Math.max(...values);
  const range = maxVal - minVal || 1;
  const width = 800;
  const height = 120;

  const points = values
    .map((v, i) => {
      const x = (i / (values.length - 1)) * (width - 40) + 40;
      const y = height - 20 - ((v - minVal) / range) * (height - 40);
      return `${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(" ");

  const thresholdY =
    threshold !== undefined && threshold >= minVal && threshold <= maxVal
      ? height - 20 - ((threshold - minVal) / range) * (height - 40)
      : null;

  return (
    <div>
      <svg viewBox={`0 0 ${width} ${height}`} className="w-full" preserveAspectRatio="none" style={{ height: "120px" }}>
        {[0, 0.5, 1].map((p) => (
          <line key={p} x1={40} y1={10 + p * 90} x2={width} y2={10 + p * 90} stroke="#e2e8f0" strokeWidth={1} />
        ))}
        {thresholdY !== null && (
          <>
            <line
              x1={40}
              y1={thresholdY}
              x2={width}
              y2={thresholdY}
              stroke="#ef4444"
              strokeWidth={1}
              strokeDasharray="4 3"
              opacity={0.5}
            />
            <text x={width - 4} y={thresholdY - 3} textAnchor="end" className="fill-destructive-400 text-[9px]">
              {thresholdLabel}
            </text>
          </>
        )}
        <polyline points={points} fill="none" stroke={color} strokeWidth={2} vectorEffect="non-scaling-stroke" />
      </svg>
      <div className="mt-2 flex justify-between text-[11px] text-secondary-400">
        <span>min {minVal.toFixed(1)}{unit}</span>
        <span>max {maxVal.toFixed(1)}{unit}</span>
      </div>
    </div>
  );
}

  return d.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit", second: "2-digit" });

import { createClient } from "npm:@supabase/supabase-js@2.57.4";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Client-Info, Apikey",
};

const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
const supabaseServiceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const supabase = createClient(supabaseUrl, supabaseServiceKey);

interface SensorPayload {
  device_id?: string;
  temperature?: number;
  humidity?: number;
  vibration_amplitude?: number;
  vibration_event?: boolean;
  temperature_warning?: boolean;
  humidity_warning?: boolean;
  total_risk?: number;
  risk_alert?: boolean;
}

function toBool(v: unknown): boolean {
  if (typeof v === "boolean") return v;
  if (typeof v === "number") return v !== 0;
  if (typeof v === "string") return v === "true" || v === "1";
  return false;
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 200, headers: corsHeaders });
  }

  if (req.method !== "POST") {
    return new Response(
      JSON.stringify({ error: "Method not allowed. Use POST." }),
      { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }

  try {
    const body: SensorPayload | SensorPayload[] = await req.json();

    const rows = Array.isArray(body) ? body : [body];

    if (rows.length === 0) {
      return new Response(
        JSON.stringify({ error: "Empty payload." }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const inserts = rows.map((row) => {
      const temperature = row.temperature ?? null;
      const humidity = row.humidity ?? null;
      const vibrationAmplitude = row.vibration_amplitude ?? null;

      const TEMP_WARNING = 35.0;
      const RH_WARNING = 65.0;
      const RH_HIGH = 75.0;
      const PIEZO_EVENT_THRESHOLD = 250;
      const RISK_ALERT_LIMIT = 30.0;

      const temperatureWarning = temperature !== null && temperature > TEMP_WARNING;
      const humidityWarning = humidity !== null && humidity > RH_WARNING;
      const humidityHigh = humidity !== null && humidity > RH_HIGH;
      const vibrationEvent = toBool(row.vibration_event) ||
        (vibrationAmplitude !== null && vibrationAmplitude >= PIEZO_EVENT_THRESHOLD);

      const espRisk = row.total_risk ?? null;
      const espAlert = toBool(row.risk_alert);
      const riskAlert = espAlert || (espRisk !== null && espRisk > RISK_ALERT_LIMIT);

      return {
        device_id: row.device_id ?? "NEUROVA-ESP32",
        temperature,
        humidity,
        vibration_amplitude: vibrationAmplitude,
        vibration_event: vibrationEvent,
        temperature_warning: temperatureWarning,
        humidity_warning: humidityWarning || humidityHigh,
        total_risk: espRisk,
        risk_alert: riskAlert,
      };
    });

    const { data, error } = await supabase
      .from("sensor_readings")
      .insert(inserts)
      .select("id, created_at");

    if (error) {
      return new Response(
        JSON.stringify({ error: error.message }),
        { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    return new Response(
      JSON.stringify({ success: true, count: data.length, entries: data }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  } catch (err) {
    return new Response(
      JSON.stringify({ error: err.message }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }
});

# Create sensor_readings table for live ESP32 data

## Purpose
Stores live grain-storage sensor readings sent from an ESP32 device
(NEUROVA Raw Paddy Rice Storage Monitor) via an edge function.

## New Tables
- `sensor_readings`
  - `id` (bigserial, primary key)
  - `device_id` (text, identifies the ESP32 device / sensor node)
  - `temperature` (real, °C from DHT22)
  - `humidity` (real, % RH from DHT22)
  - `vibration_amplitude` (integer, ADC counts from piezo)
  - `vibration_event` (boolean, true when piezo threshold crossed)
  - `temperature_warning` (boolean, true when temp > 35°C)
  - `humidity_warning` (boolean, true when RH > 65%)
  - `total_risk` (real, combined 0–100 risk score from ESP32)
  - `risk_alert` (boolean, true when total_risk > 30)
  - `created_at` (timestamptz, defaults to now())

## Security
- RLS enabled on `sensor_readings`.
- This is a single-tenant app with no sign-in screen, so anon + authenticated
  can read and insert. Updates and deletes are not needed from the frontend.
*/

  humidity real,
  vibration_amplitude integer,

  temperature_warning boolean DEFAULT false,
  humidity_warning boolean DEFAULT false,
  total_risk real,
  risk_alert boolean DEFAULT false,
  created_at timestamptz DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_sensor_readings_device

  }
}

// ================== DHT22 ==================
void readEnvironment() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();

  if (isnan(h) || isnan(t) ||
      h < 0.0 || h > 100.0 ||
      t < -40.0 || t > 80.0) {
    Serial.println("ERROR: Invalid DHT22 reading.");
    environmentValid = false;
    return;
  }
  humidity    = h;
  temperature = t;
  environmentValid = true;
  humidityWarning    = humidity > RH_WARNING;
  temperatureWarning = temperature > TEMP_WARNING;
}

// ================== PIEZO ==================
void readPiezo() {
  int minVal = 4095;
  int maxVal = 0;
  unsigned long start = micros();
  while ((unsigned long)(micros() - start) < PIEZO_WINDOW_US) {
    int v = analogRead(PIEZO_PIN);
    if (v < minVal) minVal = v;
    if (v > maxVal) maxVal = v;
  }
  piezoMinimum       = minVal;
  piezoMaximum       = maxVal;
  vibrationAmplitude = maxVal - minVal;
  vibrationRisk = clamp100(
    (float)vibrationAmplitude * 100.0 / PIEZO_FULL_SCALE);

  unsigned long now = millis();
  if (vibrationAmplitude >= PIEZO_EVENT_THRESHOLD &&
      (lastVibrationEvent == 0 ||
       now - lastVibrationEvent >= EVENT_COOLDOWN)) {
    lastVibrationEvent    = now;
    vibrationEventPending = true;
    Serial.println("VIBRATION EVENT: Threshold crossed.");
  }
}

// ================== RISK ==================
void calculateRisk() {
  if (!environmentValid) {
    totalRisk  = NAN;
    riskAlert  = false;
    return;
  }
  temperatureRisk = clamp100(
    (temperature - TEMP_BASELINE) /
    (TEMP_WARNING - TEMP_BASELINE) * 100.0);
  humidityRisk = clamp100(
    (humidity - RH_BASELINE) /
    (RH_HIGH - RH_BASELINE) * 100.0);
  totalRisk = clamp100(
    W_VIBRATION * vibrationRisk +
    W_HUMIDITY * humidityRisk +
    W_TEMPERATURE * temperatureRisk);
  riskAlert = totalRisk > RISK_ALERT_LIMIT;
}

// ================== SERIAL ==================
void printResults() {
  Serial.println("\n========== NEUROVA ==========");
  Serial.println("Material: RAW PADDY RICE");
  if (environmentValid) {
    Serial.print("Temperature: ");  Serial.print(temperature, 1); Serial.println(" C");
    Serial.print("Humidity: ");     Serial.print(humidity, 1);    Serial.println(" %");
    Serial.print("Risk: ");         Serial.print(totalRisk, 1);   Serial.println(" / 100");
    Serial.print("Alert: ");        Serial.println(riskAlert ? "ACTIVE" : "CLEAR");
  } else {
    Serial.println("ENVIRONMENT: INVALID");
  }
  Serial.print("Vibration amp: ");  Serial.print(vibrationAmplitude);
  Serial.print("  risk: ");         Serial.print(vibrationRisk, 1);
  Serial.println(" / 100");
  Serial.println("=============================");
}

// ================== SUPABASE UPLOAD ==================
void sendToSupabase() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Upload skipped: Wi-Fi disconnected.");
    return;
  }
  if (!environmentValid || isnan(totalRisk)) {
    Serial.println("Upload skipped: invalid sensor data.");
    return;
  }

  // Build JSON body with ArduinoJson
  StaticJsonDocument<256> doc;
  doc["device_id"]           = "NEUROVA-ESP32";
  doc["temperature"]         = temperature;
  doc["humidity"]            = humidity;
  doc["vibration_amplitude"] = vibrationAmplitude;
  doc["vibration_event"]     = vibrationEventPending;
  doc["temperature_warning"] = temperatureWarning;
  doc["humidity_warning"]    = humidityWarning;
  doc["total_risk"]          = totalRisk;
  doc["risk_alert"]          = riskAlert;

  String jsonBody;
  serializeJson(doc, jsonBody);

  HTTPClient http;
  http.setConnectTimeout(5000);
  http.setTimeout(5000);

  http.begin(SUPABASE_URL);
  http.addHeader("Content-Type", "application/json");

  int code = http.POST(jsonBody);

  if (code == HTTP_CODE_OK) {
    String resp = http.getString();
    Serial.print("Supabase upload OK. Entry: ");
    Serial.println(resp);
    vibrationEventPending = false;
  } else {
    Serial.print("Supabase upload failed. HTTP code: ");
    Serial.println(code);
  }

  http.end();
}

// ================== SETUP ==================
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n================================");
  Serial.println("NEUROVA - RAW PADDY MONITOR");
  Serial.println("ESP32 + DHT22 + PIEZO + SUPABASE");
  Serial.println("================================");

  analogReadResolution(12);
  analogSetPinAttenuation(PIEZO_PIN, ADC_11db);
  pinMode(PIEZO_PIN, INPUT);

  dht.begin();
  connectWiFi();

  lastDHTRead     = millis() - DHT_INTERVAL;
  lastPiezoSample = millis() - PIEZO_INTERVAL;
  lastCloudUpdate = millis() - CLOUD_INTERVAL;
  lastDisplay     = millis();
}

// ================== LOOP ==================
void loop() {
  unsigned long now = millis();

  if (WiFi.status() != WL_CONNECTED &&
      now - lastWiFiRetry >= WIFI_RETRY) {
    lastWiFiRetry = now;
    Serial.println("Retrying Wi-Fi...");
    WiFi.reconnect();
  }

  if (now - lastDHTRead >= DHT_INTERVAL) {
    lastDHTRead = now;
    readEnvironment();
  }

  now = millis();
  if (now - lastPiezoSample >= PIEZO_INTERVAL) {
    lastPiezoSample = now;
    readPiezo();
  }

  calculateRisk();

  now = millis();
  if (now - lastDisplay >= DISPLAY_INTERVAL) {
    lastDisplay = now;
    printResults();
  }

  now = millis();
  if (now - lastCloudUpdate >= CLOUD_INTERVAL) {
    lastCloudUpdate = now;
    sendToSupabase();
  }

  delay(10);
}
